Игроков в памяти: {len(USERS_DATABASE)}") except Exception as e: print(f"Ошибка чтения базы данных: {e}") USERS_DATABASE = {} else: USERS_DATABASE = {} def save_db(): try: with open(DB_FILE, "w", encoding="utf-8") as f: json.dump(USERS_DATABASE, f, ensure_ascii=False, indent=4) except Exception as e: print(f"Ошибка сохранения базы данных: {e}") def register_user(user): """Автоматически привязывает ID игрока к его имени при старте.""" changed = False if user.id not in USERS_DATABASE or USERS_DATABASE[user.id] != user.first_name: USERS_DATABASE[user.id] = user.first_name changed = True if changed: save_db() # Загружаем базу при старте load_db() # Таблица соответствия русских названий системным ключам CURRENCY_KEYS = { "доллар": "usd", "долларов": "usd", "💵": "usd", "коин": "coin", "коинов": "coin", "🪙": "coin", "мешок": "bag", "мешков": "bag", "💰": "bag", "алмаз": "diamond", "алмазов": "diamond", "💎": "diamond", "карта": "id_card", "🪪": "id_card", "медаль": "medal", "🏅": "medal", "🎖️": "s_medal", "билетик": "ticket", "🎟️": "ticket", "🔖": "vip_ticket", "диск": "disc", "💿": "disc", "📀": "g_disc", "тыковка": "pumpkin", "🎃": "pumpkin", "🍬": "candy1", "🍭": "candy2", "волна": "wave", "🌊": "wave", "мороженое": "ice", "🍦": "ice", "коктейль": "cocktail", "🍹": "cocktail", "листок": "leaf1", "🍁": "leaf1", "листья": "leaf2", "🍂": "leaf2", "яблоко": "apple", "🍏": "apple", "снежинка": "flake", "❄️": "flake", "снеговик": "snowman", "⛄": "snowman", "ёлка": "tree", "🎄": "tree", "цветок": "flower", "🌸": "flower", "цветы": "bouquet", "💐": "bouquet", } # 1. ОБРАБОТКА КОМАНДЫ /START В ЛИЧНЫХ СООБЩЕНИЯХ @bot.message_handler(commands=["start"]) def start_message(message): register_user(message.from_user) inline_markup = types.InlineKeyboardMarkup() btn_launch = types.InlineKeyboardButton( text="🚀 Начать игру", web_app=types.WebAppInfo(url=ONLINE_APP_URL) ) inline_markup.add(btn_launch) bot.send_message( message.chat.id, f"Привет, {message.from_user.first_name}! 👋\n\nЧтобы запустить игру, нажмите кнопку ниже 👇", reply_markup=inline_markup, ) # 2. НАДЕЖНАЯ АДМИН-КОМАНДА ВЫДАЧИ ПО ЦИФРОВОМУ ID ИГРАКА @bot.message_handler(commands=["add"]) def add_currency(message): # Проверка прав: доступ только у вас if message.from_user.username != ADMIN_USERNAME: bot.reply_to( message, "🛑 Отказано в доступе. Вы не являетесь создателем этой игры." ) return try: parts = message.text.split() # Проверяем количество аргументов (/add валюта количество ID) if len(parts) != 4: bot.reply_to( message, "❌ Ошибка. Пример написания команды:\n`/add доллар 500 123456789`", parse_mode="Markdown", ) return currency_name = parts[1].lower() amount_to_add = int(parts[2]) target_id_input = parts[3] # Получаем ID из команды # Проверка корректности введенного ID if not target_id_input.isdigit(): bot.reply_to(message, "❌ Ошибка: ID игрока должен состоять только из цифр!") return target_player_id = int(target_id_input) # Поиск системного ключа валюты key = CURRENCY_KEYS.get(currency_name) if not key: bot.reply_to( message, f"❌ Валюта '{currency_name}' не найдена в системе игры. Проверьте название.", ) return # Проверяем имя игрока в базе (если есть), либо подставляем заглушку player_name = USERS_DATABASE.get(target_player_id, "Новый Игрок") # Успешное начисление bot.send_message( message.chat.id, f"✅ **Успешное начисление по ID!**\n\n" f"👤 Игрок: {player_name}\n" f"🆔 ID профиля: `{target_player_id}`\n" f"💰 Валюта: `{currency_name}`\n" f"📈 Количество: `+{amount_to_add}`\n\n" f"Операция проведена. Баланс игрока обновится в Telegram Cloud Storage при входе! 🚀", parse_mode="Markdown", ) except ValueError: bot.reply_to( message, "❌ Ошибка: Количество начисляемой валюты должно быть целым числом.", ) except Exception as e: bot.reply_to(message, f"❌ Произошла непредвиденная ошибка на сервере: {e}") # 3. ПОДСТРАХОВКА НА СЛУЧАЙ ЛЮБОГО ДРУГОГО ТЕКСТА @bot.message_handler(content_types=["text"]) def handle_text_messages(message): register_user(message.from_user) if message.text.lower() in ["привет", "старт", "/start"]: start_message(message) else: inline_markup = types.InlineKeyboardMarkup() btn_launch = types.InlineKeyboardButton( text="🚀 Начать игру", web_app=types.WebAppInfo(url=ONLINE_APP_URL) ) inline_markup.add(btn_launch) bot.send_message( message.chat.id, "Чтобы запустить игру, нажмите кнопку ниже 👇", reply_markup=inline_markup, ) # ЗАПУСК СЕРВЕРА if __name__ == "__main__": print("🤖 Бот-сервер успешно обновлен!") print("🔑 Административная команда переведена на работу по ID пользователей.") bot.infinity_polling()
